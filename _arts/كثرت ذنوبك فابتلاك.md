    <button id="toggle-translation">Show translation</button>

    <div class="poem">

      <!-- Bayt 1 -->
      <div class="line">
        <div class="original">
          كثرتْ ذنوبُك فابتلاكَ محبةً<br>
          وتظنُّ يا مسكينُ زاد عذابَكْ!
        </div>
        <div class="translation">
          Your sins increased, so He tested you with love,  
          and you, poor soul, think your punishment has grown!
        </div>
      </div>

      <!-- Bayt 2 -->
      <div class="line">
        <div class="original">
          حتى إذا يوم القيامِ وجدتَهُ<br>
          قد يسَّرَ الرحمنُ ثمَّ حسابَكْ
        </div>
        <div class="translation">
          Until, on the Day of Judgment, you find  
          that the Most Merciful has made your reckoning easy.
        </div>
      </div>

      <!-- Bayt 3 -->
      <div class="line">
        <div class="original">
          ويقولُ يا عبدي صبرتَ مجاهدًا<br>
          هاتِ اليمينَ وخذْ إليكَ كتابَكْ
        </div>
        <div class="translation">
          And He says: My servant, you were patient and striving,  
          take your book in your right hand.
        </div>
      </div>

      <!-- Bayt 4 -->
      <div class="line">
        <div class="original">
          وترى جنانَ الخلدِ، أنعمْ منظرًا<br>
          يُرضَى وفيها قد سُقيتَ شرابَكْ
        </div>
        <div class="translation">
          And you will see the Gardens of Eternity — how blessed the sight,  
          where you are given your drink in full delight.
        </div>
      </div>

      <!-- Bayt 5 -->
      <div class="line">
        <div class="original">
          فيقولُ هل قد ذقتَ بؤسًا قبلَ ذا؟<br>
          فتقولُ ما بؤسُ المنالِ ثوابَكْ!
        </div>
        <div class="translation">
          He will say: Did you taste hardship before this?  
          You will reply: What hardship? The reward has erased it all!
        </div>
      </div>

      <!-- Bayt 6 -->
      <div class="line">
        <div class="original">
          فاصبرْ ولا تجزعْ، فليسَ بدائمٍ<br>
          عيشُ العناءِ، وسَلْ تجدْه أجابَكْ
        </div>
        <div class="translation">
          So be patient and do not despair,  
          for the life of hardship does not last — ask, and you will find He answered you.
        </div>
      </div>

    </div>
  </div>
</div>

document.addEventListener('DOMContentLoaded', () => {
  const button = document.getElementById('toggle-translation');
  const translations = document.querySelectorAll('.poem .translation');

  let visible = false;

  translations.forEach(t => {
    t.style.display = 'none';
  });

  button.addEventListener('click', () => {
    visible = !visible;

    translations.forEach(t => {
      t.style.display = visible ? 'block' : 'none';
    });

    button.textContent = visible ? 'Hide translation' : 'Show translation';
  });
});
</script>
