<div class="poem-page">
  <div class="poem-container">
    <button id="toggle-translation">Show translation</button>

    <div class="poem">
      <div class="line">
        <span class="original">ولما أن بدت ليلى</span>
        <span class="translation">And when Layla (symbol) appeared</span>
      </div>
      <div class="line">
        <span class="original">أخذتُ أســبحُ المولى</span>
        <span class="translation">I lifted my praises to the Lord</span>
      </div>
      <div class="line">
        <span class="original">تجلّى نورها يجلو</span>
        <span class="translation">Her radiance shone forth, revealing</span>
      </div>
      <div class="line">
        <span class="original">ظلام الليل واستولى</span>
        <span class="translation">and banishing the darkness of night</span>
      </div>
      <div class="line">
        <span class="original">على قلبي فهيمني</span>
        <span class="translation">It seized my heart entirely</span>
      </div>
      <div class="line">
        <span class="original">فلســت أرى لذا حلّا</span>
        <span class="translation">So that I could see a solution</span>
      </div>
      <div class="line">
        <span class="original">ورمتُ بأن أكلمها</span>
        <span class="translation">I longed to speak to her</span>
      </div>
      <div class="line">
        <span class="original">فلم أسـطعْ لها قولا</span>
        <span class="translation">Yet no words would come from my lips</span>
      </div>
      <div class="line">
        <span class="original">أيا ليلى أيا ليلى</span>
        <span class="translation">O Layla, O Layla</span>
      </div>
      <div class="line">
        <span class="original">سلبتِ القلبَ والعقلا</span>
        <span class="translation">You have stolen both my heart and reason</span>
      </div>
      <div class="line">
        <span class="original">أنا عبدٌ لكم يرجو</span>
        <span class="translation">I am your humble servant, yearning</span>
      </div>
      <div class="line">
        <span class="original">وصالًا فاسمحوا طولا</span>
        <span class="translation">for your embrace—grant me your grace</span>
      </div>
    </div>
  </div>
</div>

<script>
document.addEventListener('DOMContentLoaded', () => {
  const button = document.getElementById('toggle-translation');
  const translations = document.querySelectorAll('.poem .translation');

  let visible = false;

  translations.forEach(t => {
    t.style.display = 'none';
  });

  button.addEventListener('click', () => {
    visible = !visible;

    translations.forEach(t => {
      t.style.display = visible ? 'block' : 'none';
    });

    button.textContent = visible ? 'Hide translation' : 'Show translation';
  });
});
</script>
