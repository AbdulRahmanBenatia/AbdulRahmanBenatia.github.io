<div class="poem-page">
  <div class="poem">
    <button id="toggle-translation">Show translation</button>

    <div class="line">
      <span class="original">ولما أن بدت ليلى</span>
      <span class="translation">And when Layla appeared</span>
    </div>

    <div class="line">
      <span class="original">أخذتُ أســبحُ المولى</span>
      <span class="translation">I began to praise the Lord</span>
    </div>

    <div class="line">
      <span class="original">تجلّى نورها يجلو</span>
      <span class="translation">Her light shone, dispelling</span>
    </div>

    <div class="line">
      <span class="original">ظلام الليل واستولى</span>
      <span class="translation">the darkness of night and overtaking</span>
    </div>

    <div class="line">
      <span class="original">على قلبي فهيمني</span>
      <span class="translation">my heart completely</span>
    </div>

    <div class="line">
      <span class="original">فلســت أرى لذا حلّا</span>
      <span class="translation">I could not see anything else</span>
    </div>

    <div class="line">
      <span class="original">ورمتُ بأن أكلمها</span>
      <span class="translation">I wanted to speak to her</span>
    </div>

    <div class="line">
      <span class="original">فلم أسـطعْ لها قولا</span>
      <span class="translation">But I could not say a word</span>
    </div>

    <div class="line">
      <span class="original">أيا ليلى أيا ليلى</span>
      <span class="translation">O Layla, O Layla</span>
    </div>

    <div class="line">
      <span class="original">سلبتِ القلبَ والعقلا</span>
      <span class="translation">You stole my heart and reason</span>
    </div>

    <div class="line">
      <span class="original">أنا عبدٌ لكم يرجو</span>
      <span class="translation">I am your servant, hoping</span>
    </div>

    <div class="line">
      <span class="original">وصالًا فاسمحوا طولا</span>
      <span class="translation">for union—please grant me with your generousity</span>
    </div>

  </div>
</div>

  const translations = document.querySelectorAll('.poem .translation');

  translations.forEach(t => t.style.display = 'none'); // hide initially

  button.addEventListener('click', () => {
    const isVisible = translations[0].style.display === 'inline';
    translations.forEach(t => t.style.display = isVisible ? 'none' : 'inline');
    button.textContent = isVisible ? 'Show translation' : 'Hide translation';
  });
});
</script>
